package jobqueue // import "github.com/agent-fox-dev/aflib/jobqueue"

Package jobqueue provides a durable, SQLite-backed job queue with retry, per-key
serialization, crash recovery, and graceful shutdown.

CONSTANTS

const (
	StatusQueued     = "queued"
	StatusRunning    = "running"
	StatusCompleted  = "completed"
	StatusFailed     = "failed"
	StatusDeadLetter = "dead_letter"
	StatusCancelled  = "cancelled"
)
    Status constants for job lifecycle states.


VARIABLES

var ErrNotCancellable = errors.New("job is not in a cancellable status")
    ErrNotCancellable is returned by CancelJob when the target job is in a
    status that cannot be cancelled (running, completed, or dead_letter).


FUNCTIONS

func ContextWithJobID(ctx context.Context, jobID string) context.Context
    ContextWithJobID returns a new context with the given job ID set. This is
    primarily useful for testing handlers outside of the queue worker loop.

func InitSchema(db *sql.DB) error
    InitSchema creates the jobs table and required indexes using CREATE TABLE
    IF NOT EXISTS and CREATE INDEX IF NOT EXISTS. The caller's *sql.DB must
    be configured with PRAGMA journal_mode=WAL and PRAGMA busy_timeout before
    calling this function.

    InitSchema is idempotent: calling it multiple times on the same database
    produces the same schema as calling it once, and existing data is preserved.

func JobIDFromContext(ctx context.Context) string
    JobIDFromContext extracts the job ID from a context injected by the worker
    during handler dispatch. Returns an empty string if no job ID is present.

func MigrateGroupKey(db *sql.DB) error
    MigrateGroupKey adds the group_key TEXT NOT NULL DEFAULT ” column to the
    jobs table. This migration is idempotent: if the column already exists,
    it is a no-op.

    Existing rows receive group_key = ” (the column default), preserving
    backward compatibility for job types that do not use group serialization.

func MigrateProgress(db *sql.DB) error
    MigrateProgress adds the progress TEXT column to the jobs table. This
    migration is idempotent: if the column already exists, it is a no-op.

    The progress column stores intermediate JSON data written during job
    execution (e.g., rebuild patch results as they complete). Existing rows
    receive NULL.


TYPES

type EnqueueParams struct {
	Type        string
	Key         string
	Nonce       string
	Payload     json.RawMessage
	SubmittedBy string
	Group       string // Optional: when non-empty, used as group_key for group serialization.

	// ExcludeJobID, when non-empty, makes the active (type, key) duplicate
	// checks ignore the job with this ID. A running job uses it to enqueue a
	// follow-up of its own type and key, which would otherwise be
	// deduplicated against itself.
	ExcludeJobID string
}
    EnqueueParams holds the parameters for enqueueing a new job.

type HandlerFunc func(ctx context.Context, payload json.RawMessage) (result any, retryable bool, err error)
    HandlerFunc is the signature for job type handlers. On success (err == nil),
    result is serialized as JSON and stored. On failure, retryable indicates
    whether the job should be retried. Handlers must be idempotent because crash
    recovery may re-dispatch a job whose handler had already partially executed.

type Job struct {
	ID          string
	Type        string
	Key         string
	GroupKey    string // Group serialization key; empty for legacy per-key serialization.
	Nonce       string
	Status      string
	Payload     json.RawMessage
	Result      json.RawMessage
	Error       string
	Progress    json.RawMessage // Intermediate progress data written during execution.
	RetryCount  int
	AvailableAt time.Time
	SubmittedBy string
	CreatedAt   time.Time
	UpdatedAt   time.Time
}
    Job represents a persistent job record in the jobs table.

type ListOpts struct {
	Status string
	Key    string
	Offset int
	Limit  int
}
    ListOpts carries optional filter and pagination parameters for ListByType.

type Option func(*Queue) error
    Option configures the Queue constructor.

func WithGracePeriod(d time.Duration) Option
    WithGracePeriod sets the maximum duration to wait for in-flight handlers to
    complete during graceful shutdown (default 30s).

func WithPollInterval(d time.Duration) Option
    WithPollInterval sets the duration between poll cycles (default 5s).

func WithWorkers(n int) Option
    WithWorkers sets the number of worker goroutines (default 4).

type Queue struct {
	// Has unexported fields.
}
    Queue is the durable job queue backed by SQLite.

func New(db *sql.DB, logger *slog.Logger, opts ...Option) (*Queue, error)
    New creates a new Queue instance with default configuration (4 workers,
    5s poll interval, 30s grace period) overridden by any provided options. Does
    not start worker goroutines. Returns (nil, non-nil error) if configuration
    is invalid or db is nil.

func (q *Queue) CancelJob(jobID string) error
    CancelJob transitions a queued job to cancelled status. Returns nil
    on success or if the job is already cancelled (idempotent). Returns
    ErrNotCancellable for jobs in running, completed, or dead_letter status.
    Returns a not-found error if the job ID does not exist.

func (q *Queue) CountByStatus(typeName string) (map[string]int, error)
    CountByStatus returns a map of status string to integer count for all jobs
    of the given type. Statuses with zero jobs are omitted from the map.

func (q *Queue) Enqueue(params EnqueueParams) (jobID string, duplicate bool, err error)
    Enqueue enqueues a new job for execution. It validates the type is
    registered, checks nonce uniqueness, checks for active (type, key)
    duplicates, and inserts the job record. Returns (jobID, duplicate, err)
    where duplicate=true indicates a (type, key) dedup match and duplicate=false
    indicates either a new job or a nonce match.

func (q *Queue) GetByID(jobID string) (*Job, error)
    GetByID returns a single job record by its UUID.

func (q *Queue) ListByKey(typeName string, key string) ([]*Job, error)
    ListByKey returns all job records for the given (type, key) combination
    ordered by created_at descending. Returns an empty slice (not nil) when no
    jobs match.

func (q *Queue) ListByKeyPrefix(typeName, prefix string, limit int) ([]*Job, error)
    ListByKeyPrefix returns up to limit jobs of the given type whose key starts
    with prefix, ordered by created_at descending. limit <= 0 selects the
    default of 50.

func (q *Queue) ListByType(typeName string, opts ListOpts) ([]*Job, error)
    ListByType returns jobs of the given type filtered by optional status and
    key fields in opts, ordered by created_at descending, with pagination
    applied via offset and limit. Returns an empty slice (not nil) when no jobs
    match.

func (q *Queue) Register(typeName string, handler HandlerFunc, policy *RetryPolicy) error
    Register registers a job type with its handler and optional retry policy.
    Must be called before the queue is started. Returns an error if the type
    name is empty, the handler is nil, the type is already registered, or the
    queue has already been started.

func (q *Queue) RequeueDeadLetter(jobID string) (string, error)
    RequeueDeadLetter requeues a dead-lettered job for re-execution. It resets
    retry_count to 0, sets status to queued, sets available_at to now(),
    and sends a non-blocking signal on the wakeup channel. Returns (jobID, nil)
    on success. Returns (existingActiveJobID, error) if an active job exists for
    the same (type, key). Returns ("", error) if the job is not in dead_letter
    status or does not exist.

func (q *Queue) Shutdown()
    Shutdown initiates graceful shutdown by closing the stop channel to
    broadcast a stop signal to all worker goroutines. Workers stop claiming
    new jobs and complete their current handler calls. Multiple calls are safe
    (protected by sync.Once).

func (q *Queue) Start() error
    Start performs crash recovery and launches worker goroutines. Returns an
    error if crash recovery fails or if the queue is already started.

func (q *Queue) Stop() error
    Stop initiates graceful shutdown and waits for completion. Equivalent to
    calling Shutdown() followed by Wait().

func (q *Queue) UpdateProgress(jobID string, data any) error
    UpdateProgress persists serialized progress JSON to the progress column for
    a running job. The data parameter is marshalled to JSON before writing. This
    is intended to be called during handler execution to record intermediate
    progress (e.g., per-patch rebuild results).

func (q *Queue) Wait() error
    Wait blocks until all workers have exited or the grace period expires.
    If the grace period expires before all workers finish, in-flight handler
    contexts are cancelled, a WARN is logged per interrupted job, and Wait
    returns. Returns nil on clean shutdown.

type RetryPolicy struct {
	Base       time.Duration
	Multiplier float64
	Cap        time.Duration
	MaxRetries int
}
    RetryPolicy configures exponential backoff for a job type. Zero-value fields
    are replaced with defaults: Base 2s, Multiplier 2, Cap 2h, MaxRetries 20.

func DefaultRetryPolicy() RetryPolicy
    DefaultRetryPolicy returns a RetryPolicy with default values: Base=2s,
    Multiplier=2, Cap=2h, MaxRetries=20.
